set(HTTP_SOURCE ${REPO_ROOT}/lib/LuaHttp)
foreach(unit policy protocol budget)
  add_executable(LuaHttp_${unit} ${unit}_test.cpp ${HTTP_SOURCE}/HttpCore.cpp)
  target_include_directories(LuaHttp_${unit} PRIVATE ${HTTP_SOURCE})
  target_compile_options(LuaHttp_${unit} PRIVATE -UNDEBUG)
  add_test(NAME LuaHttp_${unit} COMMAND LuaHttp_${unit})
endforeach()
file(GLOB LUA_HTTP_TEST_LUA_SOURCES ${REPO_ROOT}/lib/Lua/src/l*.c)
add_library(LuaHttpTestLua STATIC ${LUA_HTTP_TEST_LUA_SOURCES})
target_include_directories(LuaHttpTestLua PUBLIC ${REPO_ROOT}/lib/Lua/src)
find_package(Threads REQUIRED)
add_executable(LuaHttp_async async_test.cpp ${HTTP_SOURCE}/HttpCore.cpp ${HTTP_SOURCE}/LuaHttp.cpp)
target_include_directories(LuaHttp_async PRIVATE ${HTTP_SOURCE})
target_link_libraries(LuaHttp_async PRIVATE LuaHttpTestLua Threads::Threads ${CMAKE_DL_LIBS} m)
target_compile_options(LuaHttp_async PRIVATE -UNDEBUG)
add_test(NAME LuaHttp_async COMMAND LuaHttp_async)
set_tests_properties(LuaHttp_async PROPERTIES ENVIRONMENT "SMUDGE_X3_CONSTRAINTS=1")
